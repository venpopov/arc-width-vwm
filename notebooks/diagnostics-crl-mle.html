NotePrerequisites

This notebook fits the Counterfactual Q-Learning model (CRL), developed in Counterfactual RL. The SDM model and optimality derivation are described in Optimal Arc Width (SDM).

Task and model summary

On each trial a participant reports a remembered color and sets a symmetric arc of half-width \(a_t \in (0,\pi)\). Points are awarded as \(p_t = (\pi - a_t)\,\mathbf{1}[|y_t| \le a_t]\). Response errors follow the SDM: \(y_t \sim \text{SDM}(c,\kappa)\).

Counterfactual Q-Learning. The agent discretizes \([0,\pi]\) into \(K\) bins \(z_1,\dots,z_K\) and maintains Q-values updated on every trial using the counterfactual reward for all actions: \[ Q_{t+1}(k) = Q_t(k) + \eta\bigl[(\pi - z_k)\,\mathbf{1}[|y_t| \le z_k] - Q_t(k)\bigr]. \]

Action selection uses a softmax policy: \[ P(\alpha_t = z_k \mid Q_t) = \frac{\exp(Q_t(k)/\tau)}{\sum_j \exp(Q_t(j)/\tau)}. \]

Joint log-likelihood. Because Q-values evolve deterministically from observed errors: \[ \ell(\theta) = \sum_t \log f_{\text{SDM}}(y_t \mid c,\kappa) + \sum_t \log P(\alpha_t \mid Q_t, \tau). \]

Parameters: \(\theta = (c, \kappa, \eta, \tau)\).

Likelihood profiles at the MLE

best_theta <- fit$par
par_names <- c("c", "kappa", "eta", "tau")

profiles <- purrr::map_dfr(seq_along(par_names), \(j) {
  grid <- seq(best_theta[j] - 1.5, best_theta[j] + 1.5, length.out = 60)
  ll_vals <- sapply(grid, \(v) {
    theta_tmp <- best_theta
    theta_tmp[j] <- v
    -neg_loglik_crl(theta_tmp, y = sim$y, a = sim$a)
  })

  if (par_names[j] == "eta") {
    value <- inv_logit(grid)
  } else {
    value <- exp(grid)
  }

  tibble(
    parameter = par_names[j],
    value = value,
    loglik = ll_vals
  )
})

Sanity-check recovery across a few parameter sets

Fit a handful of parameter combinations to verify recovery without running a full grid.

recovery_sets <- tibble(
  c     = c(2, 4, 8, 6, 10, 3),
  kappa = c(3, 3, 3, 5, 2,  8),
  eta   = c(0.05, 0.1, 0.2, 0.15, 0.1, 0.05),
  tau   = c(0.1, 0.05, 0.05, 0.1, 0.02, 0.2)
)

set.seed(2026)

fit_one_crl <- function(c, kappa, eta, tau, n_trials = 1000) {
  sim <- run_arc_algorithm(
    arc_update_fun = algo_crl(eta = eta, tau = tau),
    c = c, kappa = kappa, init_arc = pi / 2, n_trials = n_trials
  )

  theta0 <- par_to_unconstrained_crl(list(
    c = 4, kappa = 4, eta = 0.1, tau = 0.1
  ))

  fit <- tryCatch(
    optim(par = theta0, fn = neg_loglik_crl, y = sim$y, a = sim$a,
          method = "Nelder-Mead",
          control = list(maxit = 10000, reltol = 1e-10)),
    error = \(e) list(par = rep(NA_real_, 4), convergence = 99)
  )

  est <- par_from_unconstrained_crl(fit$par)
  tibble(
    c_true = c, kappa_true = kappa, eta_true = eta, tau_true = tau,
    c_est = est$c, kappa_est = est$kappa, eta_est = est$eta, tau_est = est$tau,
    converged = fit$convergence == 0
  )
}

recovery_results <- purrr::pmap_dfr(recovery_sets, fit_one_crl)

Shuffled-trial diagnostics

Shuffling trial order destroys the sequential Q-learning structure. If the model still “recovers” \(\eta\) and \(\tau\) from shuffled data, those parameters are not identified by temporal dependencies.

Single-dataset shuffle test

Shuffle the original simulated data 200 times and refit.

set.seed(2027)
n_shuffle <- 10

fit_shuffled_crl <- function(y, a) {
  idx <- sample.int(length(y))
  y_s <- y[idx]
  a_s <- a[idx]

  theta0 <- par_to_unconstrained_crl(init_pars)
  fit <- tryCatch(
    optim(par = theta0, fn = neg_loglik_crl, y = y_s, a = a_s,
          method = "Nelder-Mead",
          control = list(maxit = 10000, reltol = 1e-10)),
    error = \(e) list(par = rep(NA_real_, 4), convergence = 99)
  )

  est <- par_from_unconstrained_crl(fit$par)
  tibble(
    c = est$c, kappa = est$kappa,
    eta = est$eta, tau = est$tau,
    converged = fit$convergence == 0
  )
}

shuffle_fits <- with_cache("output/crl_shuffle_fits.rds", {
  purrr::map_dfr(seq_len(n_shuffle), \(i) fit_shuffled_crl(sim$y, sim$a))
})

shuffle_long <- shuffle_fits |>
  pivot_longer(cols = c(c, kappa, eta, tau),
               names_to = "parameter", values_to = "estimated")

ggplot(shuffle_long, aes(x = estimated)) +
  geom_histogram(bins = 30, fill = "steelblue", color = "white") +
  geom_vline(
    data = tibble(parameter = names(true_pars), true_value = unlist(true_pars)),
    aes(xintercept = true_value), color = "red", linetype = "dashed"
  ) +
  facet_wrap(~parameter, scales = "free") +
  labs(x = "Estimated value", y = "Count",
       caption = "Red dashed = generating parameter (unshuffled data)")

Arc density: original vs shuffled-fit simulations

For each shuffled fit, simulate new data from the recovered parameters and compare the arc-width density to the original data.

shuffled_sims <- purrr::pmap_dfr(
  shuffle_fits |> filter(converged),
  \(c, kappa, eta, tau, ...) {
    out <- run_arc_algorithm(
      algo_crl(eta = eta, tau = tau),
      c = c, kappa = kappa, init_arc = pi / 2, n_trials = 1000
    )
    tibble(arc = out$a)
  },
  .id = "run"
)

original_df <- tibble(arc = sim$a, source = "Original data")

ggplot() +
