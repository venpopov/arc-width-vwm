NotePrerequisites

This notebook fits the contraction-expansion model (algo4) with logit-scale noise to real experimental data using full Bayesian inference. Model diagnostics and parameter recovery are in algo4 Bayesian diagnostics. The MLE counterpart is in algo4 MLE diagnostics.

Model setup

sdm_algo4 <- custom_family(
  "sdm_algo4",
  dpars = c("mu", "c", "kappa", "delta", "asigma"),
  links = c("tan_half", "identity", "log", "logit", "log"),
  type = "real",
  loop = FALSE,
  vars = c("vreal1", "vreal2")
)

stanvars <- sdm_base_stanvars() +
  stanvar(
    scode = read_file("stan/sdm_algo4_likelihood.stan"),
    block = "functions"
  )

prior <- set_prior("constant(0)", dpar = "mu", class = "Intercept") +
  set_prior("student_t(5, 2, 0.75)", dpar = "c", class = "Intercept") +
  set_prior("student_t(5, 1.75, 0.75)", dpar = "kappa", class = "Intercept") +
  set_prior("normal(-3.5, 2)", dpar = "delta", class = "Intercept") +
  set_prior("normal(-3.5, 2)", dpar = "asigma", class = "Intercept")


fit_algo4_bayes <- function(data, file) {
  brm(
    bf(resperr | vreal(half_arc, arc_next) ~ 1, c ~ 1, kappa ~ 1, delta ~ 1, asigma ~ 1),
    prior = prior,
    data = data, family = sdm_algo4, stanvars = stanvars,
    backend = "cmdstanr", cores = 4, init = 0, file = file,
    refresh = 0
  )
}

Plot posteriors on native scale

algo4_posterior_draws <- function(fit, n_draws = NULL) {
  draws <- as_draws_df(fit) |> 
    select(starts_with("Intercept"), -Intercept)

  colnames(draws) <- gsub("Intercept_", "", colnames(draws))

  if (is.null(n_draws)) {
    idx <- seq_len(nrow(draws))
  } else {
    idx <- sample(seq_len(nrow(draws)), size = n_draws)
  }

  draws |>
    mutate(c = exp(c), kappa = exp(kappa), delta = inv_logit(delta), asigma = exp(asigma)) |> 
    slice(idx)
}

Posterior predictive check

Code
algo4_sim_from_draws <- function(draws, data) {
  purrr::pmap(
    draws,
    \(c, kappa, delta, asigma) {
      run_arc_algorithm(
        arc_update_fun = algo4_logit_noise(delta = delta, sigma_e = asigma),
        c = c,
        kappa = kappa,
        init_arc = data$half_arc[1],
        n_trials = nrow(data)
      )
    }
  )
}

plot_algo4_ppc <- function(fit, data, n_draws = 15, draws = NULL) {
  if (is.null(draws)) draws <- algo4_posterior_draws(fit, n_draws = n_draws)
  sims <- algo4_sim_from_draws(draws, data)
  posterior_col <- grDevices::adjustcolor("red", alpha.f = 0.3)

  obs_median <- median(data$half_arc)
  pred_medians <- vapply(sims, \(s) median(s$a), numeric(1))
  pred_median_mid <- median(pred_medians)
  optima <- purrr::pmap_dbl(draws, \(c, kappa, ...) find_optimum_sdm(c, kappa)$alpha)
  optimum_mid <- median(optima)

  oldpar <- par(no.readonly = TRUE)
  on.exit(par(oldpar), add = TRUE)
  par(mfrow = c(1, 2))

  plot(density(data$half_arc), main = "Arc width distribution")
  for (sim in sims) {
    lines(density(sim$a), col = posterior_col)
  }
  lines(density(data$half_arc), lwd = 2)
  abline(v = optimum_mid, col = "blue", lwd = 2, lty = 2)
  abline(v = obs_median, col = "black", lwd = 2, lty = 3)
  abline(v = pred_median_mid, col = "red", lwd = 2, lty = 3)
  legend("topright",
    c("Data", "Posterior draws", "Optimal", "Obs median", "Pred median"),
    col = c("black", "red", "blue", "black", "red"),
    lty = c(1, 1, 2, 3, 3), lwd = c(1, 1, 2, 2, 2), bty = "n")

  plot(density(data$resperr), main = "Response error distribution")
  for (sim in sims) {
    lines(density(sim$y), col = posterior_col)
  }
  lines(density(data$resperr), lwd = 2)
  legend("topright", c("Data", "Posterior draws"), col = c("black", "red"), lty = 1, bty = "n")

  invisible(sims)
}

plot_algo4_ppc_trajectories <- function(fit, data, n_draws = 15) {
  draws <- algo4_posterior_draws(fit, n_draws = n_draws)
  sims <- algo4_sim_from_draws(draws, data)

  oldpar <- par(no.readonly = TRUE)
  on.exit(par(oldpar), add = TRUE)
  par(mfrow = c(4, 4))

  plot(data$half_arc, type = "l", xlab = "Trial", ylab = "Arc half-width",
       main = "Data", ylim = c(0, pi), col = "blue")

  for (sim in sims) {
    plot(sim$a, type = "l", xlab = "Trial", ylab = "Arc half-width",
         main = "Posterior", ylim = c(0, pi))
  }

  invisible(sims)
}

summarize_algo4_posterior <- function(fit, subject_id) {
  algo4_posterior_draws(fit) |>
    pivot_longer(cols = everything(), names_to = "parameter", values_to = "est") |>
    group_by(parameter) |>
    summarise(
      median = median(est),
      lower = quantile(est, 0.025),
      upper = quantile(est, 0.975),
      .groups = "drop"
    ) |>
    mutate(subject = subject_id, .before = 1)
}

Hierarchical fit

The likelihood now handles subject boundaries via the arc_next column: the last trial of each subject has arc_next = -1, so the Stan code skips the arc-transition term for those rows. This makes it safe to stack all subjects’ data.

hfit <- brm(
  bf(
    resperr | vreal(half_arc, arc_next) ~ 1,
    c ~ 1 + (1 | subject),
    kappa ~ 1 + (1 | subject),

Extract subject-level posterior draws

We use coef() to obtain posterior draws of the subject-level parameters (population intercept + subject-specific random effect), then transform them to the native scale.

Code
algo4_hier_subject_draws <- function(hfit, subject_id, n_draws = NULL) {
  coefs <- coef(hfit, summary = FALSE)
  idx <- if (is.null(n_draws)) TRUE else sample(nrow(coefs$subject[, , 1]), n_draws)

  tibble(
    c     = exp(coefs$subject[idx, subject_id, "c_Intercept"]),
    kappa = exp(coefs$subject[idx, subject_id, "kappa_Intercept"]),
    delta = inv_logit(coefs$subject[idx, subject_id, "delta_Intercept"]),
    asigma = exp(coefs$subject[idx, subject_id, "asigma_Intercept"])
  )
}

Population-level posterior draws

Code
algo4_hier_population_draws <- function(hfit, n_draws = NULL) {
  draws <- as_draws_df(hfit) |>
    select(b_c_Intercept, b_kappa_Intercept, b_delta_Intercept, b_asigma_Intercept)

  idx <- if (is.null(n_draws)) seq_len(nrow(draws)) else sample(nrow(draws), n_draws)

  tibble(
    c     = exp(draws$b_c_Intercept[idx]),
    kappa = exp(draws$b_kappa_Intercept[idx]),
    delta = inv_logit(draws$b_delta_Intercept[idx]),
    asigma = exp(draws$b_asigma_Intercept[idx])
  )
}

hier_ppc_overall <- function(hfit, data, n_draws = 15) {
